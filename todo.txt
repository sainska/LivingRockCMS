CREATE TABLE households (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name VARCHAR(100),

  created_at TIMESTAMP DEFAULT NOW()

🛡️ 3. roles
CREATE TABLE roles (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name VARCHAR(50) UNIQUE, -- e.g., Member, Admin, Secretary, Treasurer

🔐 4. user_roles
CREATE TABLE user_roles (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES users(id) ON DELETE CASCADE,
  role_id UUID REFERENCES roles(id),
  assigned_by UUID REFERENCES users(id),
  assigned_at TIMESTAMP DEFAULT NOW()
);

CREATE TABLE events (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  title VARCHAR(100),
  description TEXT,
  start_datetime TIMESTAMP,

  location TEXT,
  created_by UUID REFERENCES users(id),
  type VARCHAR(50), -- e.g., Worship, Conference, Training
  created_at TIMESTAMP DEFAULT NOW()
);
📝 6. event_registrations

  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

  user_id UUID REFERENCES users(id) ON DELETE CASCADE,
  registered_at TIMESTAMP DEFAULT NOW(),
  attendance_status VARCHAR(20) DEFAULT 'pending' -- present, absent
);
💰 7. donations
CREATE TABLE donations (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

  donation_type VARCHAR(50), -- e.g., Offering, Tithe, Pledge
  receipt_url TEXT,
  transaction_reference VARCHAR(100),
  donated_at TIMESTAMP DEFAULT NOW()
);
📈 8. pledges
CREATE TABLE pledges (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES users(id),

  fulfilled BOOLEAN DEFAULT FALSE,
  created_at TIMESTAMP DEFAULT NOW()
);
💸 9. expenses
CREATE TABLE expenses (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

  receipt_url TEXT,
  spent_at TIMESTAMP DEFAULT NOW()
);
🧾 10. attendance_logs
CREATE TABLE attendance_logs (
